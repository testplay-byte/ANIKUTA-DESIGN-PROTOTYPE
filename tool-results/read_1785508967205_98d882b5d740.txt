     1→// wizard_visuals.dart — custom-painted and composed animations for each screen.
     2→//
     3→// Each visual is a self-contained stateless widget that takes the current
     4→// ColorScheme (or palette) so it adapts to the selected theme. Animations use
     5→// built-in AnimationController + CustomPainter (zero extra dependencies).

    29→class _WelcomeVisualState extends State<WelcomeVisual>
    30→    with TickerProviderStateMixin {
    31→  late final AnimationController _c;
    32→
    33→  @override
    34→  void initState() {
    35→    super.initState();
    36→    _c = AnimationController(

    70→class _WelcomePainter extends CustomPainter {
    71→  final Color primary;
    72→  final Color onPrimary;
    73→  final double scale;
    74→  _WelcomePainter({
    75→    required this.primary,
    76→    required this.onPrimary,
    77→    required this.scale,
    78→  });
    79→
    80→  @override
    81→  void paint(Canvas canvas, Size size) {
    82→    final cx = size.width / 2;
    83→    final cy = size.height / 2;
    84→    // Soft glow
    85→    final glowPaint = Paint()..color = primary.withOpacity(0.28);
    86→    canvas.drawCircle(Offset(cx, cy), size.width * 0.42, glowPaint);
    87→    // Rounded square
    88→    final sqSize = size.width * 0.42 * scale;
    89→    final rrect = RRect.fromRectAndRadius(
    90→      Rect.fromCenter(center: Offset(cx, cy), width: sqSize, height: sqSize),
    91→      Radius.circular(sqSize * 0.26),
    92→    );
    93→    canvas.drawRRect(rrect, Paint()..color = primary);
    94→    // Play triangle
    95→    final tri = Path()
    96→      ..moveTo(cx - sqSize * 0.10, cy - sqSize * 0.19)
    97→      ..lineTo(cx - sqSize * 0.10, cy + sqSize * 0.19)
    98→      ..lineTo(cx + sqSize * 0.21, cy)
    99→      ..close();
   100→    canvas.drawPath(tri, Paint()..color = onPrimary);
   101→  }
   102→
   103→  @override
   104→  bool shouldRepaint(covariant _WelcomePainter old) =>
   105→      old.scale != scale || old.primary != primary;
   106→}

   179→class _FolderPainter extends CustomPainter {
   180→  final Color primary;
   181→  final Color surface3;
   182→  final Color surface4;
   183→  final Color surface5;
   184→  final Color background;
   185→  final bool selected;
   186→  final double bob; // 0..1
   187→  _FolderPainter({
   188→    required this.primary,
   189→    required this.surface3,
   190→    required this.surface4,
   191→    required this.surface5,
   192→    required this.background,
   193→    required this.selected,
   194→    required this.bob,
   195→  });
   196→
   197→  @override
   198→  void paint(Canvas canvas, Size size) {
   199→    final u = size.width / 200;
   200→    final cx = size.width / 2;
   201→    // glow
   202→    canvas.drawCircle(
   203→        Offset(cx, 140 * u), 60 * u, Paint()..color = primary.withOpacity(0.18));
   204→    final dy = (bob - 0.5) * 8 * u; // -4u..+4u
   205→    // Back lid
   206→    _rr(canvas, surface4, 40 * u, (88 + dy) * u, 118 * u, 28 * u, 8 * u);
   207→    // Files (floating, staggered)
   208→    _file(canvas, surface5, primary, 56 * u, (56 + dy * 0.5) * u, 28 * u, 34 * u, 3 * u);
   209→    _file(canvas, surface4, primary, 86 * u, (50 + dy * 0.6) * u, 28 * u, 40 * u, 3 * u);
   210→    _file(canvas, surface3, primary, 116 * u, (56 + dy * 0.4) * u, 28 * u, 34 * u, 3 * u);
   211→    // Front pocket
   212→    _rr(canvas, surface3, 38 * u, 116 * u, 124 * u, 58 * u, 8 * u);
   213→    _rrStroke(canvas, surface5, 38 * u, 116 * u, 124 * u, 58 * u, 8 * u, 2.2 * u);
   214→    // inner lines
   215→    canvas.drawRect(
   216→        Rect.fromLTWH(54 * u, 134 * u, 92 * u, 4 * u),
   217→        Paint()..color = primary.withOpacity(0.35));
   218→    canvas.drawRect(
   219→        Rect.fromLTWH(54 * u, 148 * u, 72 * u, 4 * u),
   220→        Paint()..color = primary.withOpacity(0.25));
   221→    canvas.drawRect(
   222→        Rect.fromLTWH(54 * u, 162 * u, 82 * u, 4 * u),
   223→        Paint()..color = primary.withOpacity(0.25));
   224→    // selected check badge
   225→    if (selected) {
   226→      canvas.drawCircle(Offset(150 * u, 92 * u), 20 * u, Paint()..color = primary);
   227→      final p = Paint()
   228→        ..color = background
   229→        ..style = PaintingStyle.stroke
   230→        ..strokeWidth = 3.5 * u
   231→        ..strokeCap = StrokeCap.round;
   232→      canvas.drawLine(Offset(141 * u, 92 * u), Offset(148 * u, 99 * u), p);
   233→      canvas.drawLine(Offset(148 * u, 99 * u), Offset(160 * u, 85 * u), p);
   234→    }
   235→  }
   236→
   237→  void _rr(Canvas c, Color color, double x, double y, double w, double h, double r) {
   238→    c.drawRRect(
   239→      RRect.fromRectAndRadius(Rect.fromLTWH(x, y, w, h), Radius.circular(r)),
   240→      Paint()..color = color,
   241→    );
   242→  }
   243→
   244→  void _rrStroke(Canvas c, Color color, double x, double y, double w, double h,
   245→      double r, double stroke) {
   246→    c.drawRRect(
   247→      RRect.fromRectAndRadius(Rect.fromLTWH(x, y, w, h), Radius.circular(r)),
   248→      Paint()
   249→        ..color = color
   250→        ..style = PaintingStyle.stroke
   251→        ..strokeWidth = stroke,
   252→    );
   253→  }
   254→
   255→  void _file(Canvas c, Color body, Color accent, double x, double y, double w,
   256→      double h, double r) {
   257→    _rr(c, body, x, y, w, h, r);
   258→    c.drawRRect(
   259→      RRect.fromRectAndRadius(Rect.fromLTWH(x, y, w, h * 0.29), Radius.circular(r)),
   260→      Paint()..color = accent.withOpacity(0.5),
   261→    );
   262→  }
   263→
   264→  @override
   265→  bool shouldRepaint(covariant _FolderPainter old) =>
   266→      old.selected != selected || old.bob != bob || old.primary != primary;
   267→}

   328→class _ShieldPainter extends CustomPainter {
   329→  final Color primary;
   330→  final Color onPrimary;
   331→  final double t; // 0..1 looping
   332→  _ShieldPainter({required this.primary, required this.onPrimary, required this.t});
   333→
   334→  @override
   335→  void paint(Canvas canvas, Size size) {
   336→    final cx = size.width / 2;
   337→    final cy = size.height / 2;
   338→    // ripple
   339→    for (int i = 0; i < 2; i++) {
   340→      final phase = (t + i * 0.5) % 1.0;
   341→      final r = size.width * (0.30 + 0.22 * phase);
   342→      canvas.drawCircle(
   343→        Offset(cx, cy),
   344→        r,
   345→        Paint()
   346→          ..color = primary.withOpacity((1 - phase) * 0.25)
   347→          ..style = PaintingStyle.stroke
   348→          ..strokeWidth = 2,
   349→      );
   350→    }
   351→    // shield body
   352→    final w = size.width * 0.42;
   353→    final h = size.width * 0.50;
   354→    final path = Path()
   355→      ..moveTo(cx, cy - h / 2)
   356→      ..lineTo(cx + w / 2, cy - h / 2 + w * 0.3)
   357→      ..lineTo(cx + w / 2, cy + h * 0.1)
   358→      ..quadraticBezierTo(cx + w / 2, cy + h * 0.45, cx, cy + h / 2)
   359→      ..quadraticBezierTo(cx - w / 2, cy + h * 0.45, cx - w / 2, cy + h * 0.1)
   360→      ..lineTo(cx - w / 2, cy - h / 2 + w * 0.3)
   361→      ..close();
   362→    canvas.drawPath(path, Paint()..color = primary);
   363→    // check
   364→    final p = Paint()
   365→      ..color = onPrimary
   366→      ..style = PaintingStyle.stroke
   367→      ..strokeWidth = size.width * 0.045
   368→      ..strokeCap = StrokeCap.round
   369→      ..strokeJoin = StrokeJoin.round;
   370→    canvas.drawLine(
   371→        Offset(cx - w * 0.14, cy), Offset(cx - w * 0.02, cy + h * 0.12), p);
   372→    canvas.drawLine(
   373→        Offset(cx - w * 0.02, cy + h * 0.12), Offset(cx + w * 0.18, cy - h * 0.12), p);
   374→  }
   375→
   376→  @override
   377→  bool shouldRepaint(covariant _ShieldPainter old) =>
   378→      old.t != t || old.primary != primary;
   379→}

   443→class _RestorePainter extends CustomPainter {
   444→  final Color primary;
   445→  final Color onPrimary;
   446→  final Color surface;
   447→  final double t;
   448→  _RestorePainter({
   449→    required this.primary,
   450→    required this.onPrimary,
   451→    required this.surface,
   452→    required this.t,
   453→  });
   454→
   455→  @override
   456→  void paint(Canvas canvas, Size size) {
   457→    final cx = size.width / 2;
   458→    final cy = size.height / 2;
   459→    // file card
   460→    final w = size.width * 0.46;
   461→    final h = size.width * 0.56;
   462→    final card = RRect.fromRectAndRadius(
   463→      Rect.fromCenter(center: Offset(cx, cy), width: w, height: h),
   464→      Radius.circular(w * 0.12),
   465→    );
   466→    canvas.drawRRect(card, Paint()..color = surface);
   467→    // file fold corner
   468→    final fold = Path()
   469→      ..moveTo(cx + w / 2 - w * 0.22, cy - h / 2)
   470→      ..lineTo(cx + w / 2, cy - h / 2 + w * 0.22)
   471→      ..lineTo(cx + w / 2 - w * 0.22, cy - h / 2 + w * 0.22)
   472→      ..close();
   473→    canvas.drawPath(fold, Paint()..color = primary.withOpacity(0.5));
   474→    // content lines on card
   475→    for (int i = 0; i < 3; i++) {
   476→      canvas.drawRect(
   477→        Rect.fromLTWH(
   478→            cx - w * 0.28, cy - h * 0.08 + i * h * 0.14, w * 0.56, h * 0.04),
   479→        Paint()..color = primary.withOpacity(0.3),
   480→      );
   481→    }
   482→    // rotating restore arrow (arc + arrowhead) around the card
   483→    final ringR = size.width * 0.42;
   484→    final sweep = 1.4; // radians of arc
   485→    final start = -math.pi / 2 + t * 2 * math.pi;
   486→    canvas.drawArc(
   487→      Rect.fromCircle(center: Offset(cx, cy), radius: ringR),
   488→      start,
   489→      sweep,
   490→      false,
   491→      Paint()
   492→        ..color = primary
   493→        ..style = PaintingStyle.stroke
   494→        ..strokeWidth = size.width * 0.035
   495→        ..strokeCap = StrokeCap.round,
   496→    );
   497→    // arrowhead at end of arc
   498→    final endAngle = start + sweep;
   499→    final tip = Offset(cx + ringR * math.cos(endAngle), cy + ringR * math.sin(endAngle));
   500→    final ahead = Offset(
   501→        cx + (ringR - 10) * math.cos(endAngle), cy + (ringR - 10) * math.sin(endAngle));
   502→    final p = Paint()
   503→      ..color = primary
   504→      ..style = PaintingStyle.fill;
   505→    canvas.drawCircle(tip, size.width * 0.03, p);
   506→    canvas.drawLine(
   507→        tip, ahead, Paint()..color = primary..strokeWidth = size.width * 0.035..strokeCap = StrokeCap.round);
   508→  }
   509→
   510→  @override
   511→  bool shouldRepaint(covariant _RestorePainter old) =>
   512→      old.t != t || old.primary != primary;
   513→}

   668→class _ProcessingPainter extends CustomPainter {
   669→  final Color primary;
   670→  final Color surface;
   671→  final double t;
   672→  _ProcessingPainter({required this.primary, required this.surface, required this.t});
   673→
   674→  @override
   675→  void paint(Canvas canvas, Size size) {
   676→    final cx = size.width / 2;
   677→    final w = size.width * 0.66;
   678→    final rowH = size.width * 0.10;
   679→    final gap = size.width * 0.03;
   680→    final rows = 4;
   681→    final totalH = rows * rowH + (rows - 1) * gap;
   682→    final top = (size.height - totalH) / 2;
   683→    for (int i = 0; i < rows; i++) {
   684→      final y = top + i * (rowH + gap);
   685→      final reveal = ((t * rows) - i).clamp(0.0, 1.0);
   686→      final filled = reveal * w;
   687→      canvas.drawRRect(
   688→        RRect.fromRectAndRadius(
   689→          Rect.fromLTWH(cx - w / 2, y, w, rowH),
   690→          Radius.circular(rowH * 0.3),
   691→        ),
   692→        Paint()..color = surface,
   693→      );
   694→      if (filled > 0) {
   695→        canvas.drawRRect(
   696→          RRect.fromRectAndRadius(
   697→            Rect.fromLTWH(cx - w / 2, y, filled, rowH),
   698→            Radius.circular(rowH * 0.3),
   699→          ),
   700→          Paint()..color = primary.withOpacity(0.85),
   701→        );
   702→      }
   703→    }
   704→  }
   705→
   706→  @override
   707→  bool shouldRepaint(covariant _ProcessingPainter old) =>
   708→      old.t != t || old.primary != primary;
   709→}

   773→class _CheckCirclePainter extends CustomPainter {
   774→  final Color primary;
   775→  final Color onPrimary;
   776→  final double progress; // 0..1
   777→  final bool confetti;
   778→  _CheckCirclePainter({
   779→    required this.primary,
   780→    required this.onPrimary,
   781→    required this.progress,
   782→    required this.confetti,
   783→  });
   784→
   785→  @override
   786→  void paint(Canvas canvas, Size size) {
   787→    final cx = size.width / 2;
   788→    final cy = size.height / 2;
   789→    final r = size.width * 0.36;
   790→    // glow
   791→    canvas.drawCircle(
   792→        Offset(cx, cy), r * 1.25, Paint()..color = primary.withOpacity(0.22));
   793→    // circle (draws in)
   794→    final circleSweep = (progress * 1.2).clamp(0.0, 1.0);
   795→    canvas.drawArc(
   796→      Rect.fromCircle(center: Offset(cx, cy), radius: r),
   797→      -math.pi / 2,
   798→      circleSweep * 2 * math.pi,
   799→      false,
   800→      Paint()
   801→        ..color = primary
   802→        ..style = PaintingStyle.stroke
   803→        ..strokeWidth = size.width * 0.05
   804→        ..strokeCap = StrokeCap.round,
   805→    );
   806→    // fill once drawn
   807→    if (circleSweep >= 1.0) {
   808→      canvas.drawCircle(Offset(cx, cy), r, Paint()..color = primary);
   809→    }
   810→    // check (draws after circle)
   811→    final checkP = (progress * 1.6 - 0.6).clamp(0.0, 1.0);
   812→    if (checkP > 0) {
   813→      final p1 = Offset(cx - r * 0.30, cy);
   814→      final p2 = Offset(cx - r * 0.05, cy + r * 0.28);
   815→      final p3 = Offset(cx + r * 0.34, cy - r * 0.28);
   816→      final mid = Offset(p1.dx + (p2.dx - p1.dx) * checkP.clamp(0.0, 0.5) * 2,
   817→          p1.dy + (p2.dy - p1.dy) * checkP.clamp(0.0, 0.5) * 2);
   818→      final paint = Paint()
   819→        ..color = onPrimary
   820→        ..style = PaintingStyle.stroke
   821→        ..strokeWidth = size.width * 0.06
   822→        ..strokeCap = StrokeCap.round
   823→        ..strokeJoin = StrokeJoin.round;
   824→      if (checkP <= 0.5) {
   825→        canvas.drawLine(p1, mid, paint);
   826→      } else {
   827→        canvas.drawLine(p1, p2, paint);
   828→        final k = (checkP - 0.5) * 2;
   829→        canvas.drawLine(p2, Offset(p2.dx + (p3.dx - p2.dx) * k, p2.dy + (p3.dy - p2.dy) * k), paint);
   830→      }
   831→    }
   832→    // confetti
   833→    if (confetti && progress > 0.6) {
   834→      final rng = math.Random(42);
   835→      for (int i = 0; i < 18; i++) {
   836→        final a = rng.nextDouble() * 2 * math.pi;
   837→        final dist = r * (1.3 + 0.5 * rng.nextDouble()) * progress;
   838→        final dx = cx + dist * math.cos(a);
   839→        final dy = cy + dist * math.sin(a);
   840→        canvas.drawCircle(
   841→            Offset(dx, dy), size.width * 0.012, Paint()..color = primary.withOpacity(0.6));
   842→      }
   843→    }
   844→  }
   845→
   846→  @override
   847→  bool shouldRepaint(covariant _CheckCirclePainter old) =>
   848→      old.progress != progress || old.primary != primary;
   849→}
