     1→// wizard_visuals.dart — high-quality custom-painted animations (v2).
     2→//
     3→// Every visual is a self-contained StatefulWidget that takes the current
     4→// palette colors so it adapts to the selected theme. All animations use
     5→// built-in AnimationController + CustomPainter (zero extra dependencies).
     6→// Each is wrapped in RepaintBoundary by the WizardScaffold for smoothness.

    81→class _WelcomePainter extends CustomPainter {
    82→  final Color primary;
    83→  final Color onPrimary;
    84→  final double pulse; // 0..1 breathing
    85→  final double orbit; // 0..1 rotating
    86→  _WelcomePainter({
    87→    required this.primary,
    88→    required this.onPrimary,
    89→    required this.pulse,
    90→    required this.orbit,
    91→  });
    92→
    93→  @override
    94→  void paint(Canvas canvas, Size size) {
    95→    final cx = size.width / 2;
    96→    final cy = size.height / 2;
    97→    final s = size.width;
    98→
    99→    // Outer glow — breathes
   100→    final glowR = s * (0.42 + 0.06 * pulse);
   101→    final glowAlpha = 0.18 + 0.12 * pulse;
   102→    final glowPaint = Paint()
   103→      ..shader = RadialGradient(
   104→        colors: [primary.withOpacity(glowAlpha), primary.withOpacity(0)],
   105→      ).createShader(Rect.fromCircle(center: Offset(cx, cy), radius: glowR));
   106→    canvas.drawCircle(Offset(cx, cy), glowR, glowPaint);
   107→
   108→    // 3 orbiting accent dots
   109→    for (int i = 0; i < 3; i++) {
   110→      final angle = orbit * 2 * math.pi + (i * 2 * math.pi / 3);
   111→      final orbR = s * 0.40;
   112→      final dx = cx + orbR * math.cos(angle);
   113→      final dy = cy + orbR * math.sin(angle);
   114→      final dotR = s * 0.018;
   115→      canvas.drawCircle(
   116→          Offset(dx, dy), dotR, Paint()..color = primary.withOpacity(0.7));
   117→    }
   118→
   119→    // Main rounded-square logo — gentle scale pulse
   120→    final logoScale = 0.94 + 0.06 * pulse;
   121→    final logoSize = s * 0.40 * logoScale;
   122→    final logoR = logoSize * 0.28;
   123→    final logoPaint = Paint()..color = primary;
   124→    canvas.drawRRect(
   125→      RRect.fromRectAndRadius(
   126→        Rect.fromCenter(
   127→            center: Offset(cx, cy), width: logoSize, height: logoSize),
   128→        Radius.circular(logoR),
   129→      ),
   130→      logoPaint,
   131→    );
   132→
   133→    // Play triangle inside
   134→    final tri = Path()
   135→      ..moveTo(cx - logoSize * 0.10, cy - logoSize * 0.20)
   136→      ..lineTo(cx - logoSize * 0.10, cy + logoSize * 0.20)
   137→      ..lineTo(cx + logoSize * 0.22, cy)
   138→      ..close();
   139→    canvas.drawPath(tri, Paint()..color = onPrimary);
   140→  }
   141→
   142→  @override
   143→  bool shouldRepaint(covariant _WelcomePainter old) =>
   144→      old.pulse != pulse || old.orbit != orbit || old.primary != primary;
   145→}

   220→class _FolderPainter extends CustomPainter {
   221→  final Color primary;
   222→  final Color surface3;
   223→  final Color surface4;
   224→  final Color surface5;
   225→  final Color background;
   226→  final bool selected;
   227→  final double t; // 0..1 float
   228→  _FolderPainter({
   229→    required this.primary,
   230→    required this.surface3,
   231→    required this.surface4,
   232→    required this.surface5,
   233→    required this.background,
   234→    required this.selected,
   235→    required this.t,
   236→  });
   237→
   238→  @override
   239→  void paint(Canvas canvas, Size size) {
   240→    final u = size.width / 200;
   241→    final cx = size.width / 2;
   242→    final cy = size.height / 2;
   243→
   244→    // Soft glow behind folder
   245→    final glowPaint = Paint()
   246→      ..shader = RadialGradient(
   247→        colors: [primary.withOpacity(0.16), primary.withOpacity(0)],
   248→      ).createShader(Rect.fromCircle(center: Offset(cx, cy + 20 * u), radius: 80 * u));
   249→    canvas.drawCircle(Offset(cx, cy + 20 * u), 80 * u, glowPaint);
   250→
   251→    // Floating offset (subtle up/down)
   252→    final dy = (t - 0.5) * 6 * u;
   253→
   254→    // --- 3 floating file cards above the folder (staggered) ---
   255→    final fileW = 30 * u;
   256→    final fileH = 38 * u;
   257→    final fileR = 4 * u;
   258→    for (int i = 0; i < 3; i++) {
   259→      final fx = (62 + i * 38) * u;
   260→      final fy = (52 + dy * (0.6 + i * 0.2) + i * 2) * u;
   261→      // Card body with gradient
   262→      final cardPaint = Paint()
   263→        ..shader = LinearGradient(
   264→          begin: Alignment.topCenter,
   265→          end: Alignment.bottomCenter,
   266→          colors: [surface5, surface4],
   267→        ).createShader(Rect.fromLTWH(fx, fy, fileW, fileH));
   268→      canvas.drawRRect(
   269→        RRect.fromRectAndRadius(Rect.fromLTWH(fx, fy, fileW, fileH), Radius.circular(fileR)),
   270→        cardPaint,
   271→      );
   272→      // Colored top strip
   273→      canvas.drawRRect(
   274→        RRect.fromRectAndRadius(
   275→          Rect.fromLTWH(fx, fy, fileW, 9 * u),
   276→          Radius.circular(fileR),
   277→        ),
   278→        Paint()..color = primary.withOpacity(0.55 - i * 0.08),
   279→      );
   280→      // Content lines on file
   281→      for (int j = 0; j < 2; j++) {
   282→        canvas.drawRect(
   283→          Rect.fromLTWH(fx + 5 * u, fy + 14 * u + j * 7 * u, fileW - 10 * u, 2 * u),
   284→          Paint()..color = primary.withOpacity(0.25),
   285→        );
   286→      }
   287→    }
   288→
   289→    // --- Folder back panel (the tab part) ---
   290→    final backPath = Path()
   291→      ..moveTo(36 * u, 88 * u + dy)
   292→      ..lineTo(80 * u, 88 * u + dy)
   293→      ..lineTo(88 * u, 80 * u + dy)
   294→      ..lineTo(164 * u, 80 * u + dy)
   295→      ..lineTo(164 * u, 116 * u + dy)
   296→      ..lineTo(36 * u, 116 * u + dy)
   297→      ..close();
   298→    canvas.drawPath(backPath, Paint()..color = surface4);
   299→
   300→    // --- Folder front pocket (gradient body) ---
   301→    final frontPaint = Paint()
   302→      ..shader = LinearGradient(
   303→        begin: Alignment.topCenter,
   304→        end: Alignment.bottomCenter,
   305→        colors: [surface3, surface5],
   306→      ).createShader(Rect.fromLTWH(32 * u, 110 * u + dy, 136 * u, 64 * u));
   307→    canvas.drawRRect(
   308→      RRect.fromRectAndRadius(
   309→        Rect.fromLTWH(32 * u, 110 * u + dy, 136 * u, 64 * u),
   310→        Radius.circular(10 * u),
   311→      ),
   312→      frontPaint,
   313→    );
   314→
   315→    // Subtle primary border on folder
   316→    canvas.drawRRect(
   317→      RRect.fromRectAndRadius(
   318→        Rect.fromLTWH(32 * u, 110 * u + dy, 136 * u, 64 * u),
   319→        Radius.circular(10 * u),
   320→      ),
   321→      Paint()
   322→        ..color = primary.withOpacity(0.25)
   323→        ..style = PaintingStyle.stroke
   324→        ..strokeWidth = 1.5 * u,
   325→    );
   326→
   327→    // Inner content lines on folder
   328→    for (int i = 0; i < 3; i++) {
   329→      canvas.drawRRect(
   330→        RRect.fromRectAndRadius(
   331→          Rect.fromLTWH(50 * u, (124 + i * 12) * u, 100 * u, 4 * u),
   332→          Radius.circular(2 * u),
   333→        ),
   334→        Paint()..color = primary.withOpacity(0.30 - i * 0.06),
   335→      );
   336→    }
   337→
   338→    // --- Selected check badge ---
   339→    if (selected) {
   340→      final badgeCx = 156 * u;
   341→      final badgeCy = 102 * u + dy;
   342→      final badgeR = 18 * u;
   343→      // Badge glow
   344→      canvas.drawCircle(
   345→        Offset(badgeCx, badgeCy),
   346→        badgeR + 4 * u,
   347→        Paint()..color = primary.withOpacity(0.3),
   348→      );
   349→      // Badge circle
   350→      canvas.drawCircle(
   351→        Offset(badgeCx, badgeCy),
   352→        badgeR,
   353→        Paint()..color = primary,
   354→      );
   355→      // Check mark
   356→      final checkPaint = Paint()
   357→        ..color = background
   358→        ..style = PaintingStyle.stroke
   359→        ..strokeWidth = 3 * u
   360→        ..strokeCap = StrokeCap.round
   361→        ..strokeJoin = StrokeJoin.round;
   362→      canvas.drawLine(
   363→        Offset(badgeCx - 7 * u, badgeCy),
   364→        Offset(badgeCx - 2 * u, badgeCy + 5 * u),
   365→        checkPaint,
   366→      );
   367→      canvas.drawLine(
   368→        Offset(badgeCx - 2 * u, badgeCy + 5 * u),
   369→        Offset(badgeCx + 7 * u, badgeCy - 5 * u),
   370→        checkPaint,
   371→      );
   372→    }
   373→  }
   374→
   375→  @override
   376→  bool shouldRepaint(covariant _FolderPainter old) =>
   377→      old.selected != selected || old.t != t || old.primary != primary;
   378→}

   446→class _ShieldPainter extends CustomPainter {
   447→  final Color primary;
   448→  final Color onPrimary;
   449→  final double ripple; // 0..1
   450→  final double draw; // 0..1 check draw
   451→  _ShieldPainter({
   452→    required this.primary,
   453→    required this.onPrimary,
   454→    required this.ripple,
   455→    required this.draw,
   456→  });
   457→
   458→  @override
   459→  void paint(Canvas canvas, Size size) {
   460→    final cx = size.width / 2;
   461→    final cy = size.height / 2;
   462→    final s = size.width;
   463→
   464→    // Expanding ripple rings
   465→    for (int i = 0; i < 2; i++) {
   466→      final phase = (ripple + i * 0.5) % 1.0;
   467→      final r = s * (0.32 + 0.22 * phase);
   468→      canvas.drawCircle(
   469→        Offset(cx, cy),
   470→        r,
   471→        Paint()
   472→          ..color = primary.withOpacity((1 - phase) * 0.20)
   473→          ..style = PaintingStyle.stroke
   474→          ..strokeWidth = 2,
   475→      );
   476→    }
   477→
   478→    // Shield glow
   479→    canvas.drawCircle(
   480→      Offset(cx, cy),
   481→      s * 0.38,
   482→      Paint()
   483→        ..shader = RadialGradient(
   484→          colors: [primary.withOpacity(0.15), primary.withOpacity(0)],
   485→        ).createShader(Rect.fromCircle(center: Offset(cx, cy), radius: s * 0.38)),
   486→    );
   487→
   488→    // Shield body
   489→    final w = s * 0.40;
   490→    final h = s * 0.48;
   491→    final shieldPath = Path()
   492→      ..moveTo(cx, cy - h / 2)
   493→      ..lineTo(cx + w / 2, cy - h / 2 + w * 0.35)
   494→      ..lineTo(cx + w / 2, cy + h * 0.10)
   495→      ..quadraticBezierTo(cx + w / 2, cy + h * 0.42, cx, cy + h / 2)
   496→      ..quadraticBezierTo(cx - w / 2, cy + h * 0.42, cx - w / 2, cy + h * 0.10)
   497→      ..lineTo(cx - w / 2, cy - h / 2 + w * 0.35)
   498→      ..close();
   499→
   500→    // Shield fill with gradient
   501→    final shieldPaint = Paint()
   502→      ..shader = LinearGradient(
   503→        begin: Alignment.topCenter,
   504→        end: Alignment.bottomCenter,
   505→        colors: [primary, primary.withOpacity(0.8)],
   506→      ).createShader(Rect.fromCircle(center: Offset(cx, cy), radius: s * 0.3));
   507→    canvas.drawPath(shieldPath, shieldPaint);
   508→
   509→    // Drawing check mark (animated)
   510→    final checkPaint = Paint()
   511→      ..color = onPrimary
   512→      ..style = PaintingStyle.stroke
   513→      ..strokeWidth = s * 0.04
   514→      ..strokeCap = StrokeCap.round
   515→      ..strokeJoin = StrokeJoin.round;
   516→    final p1 = Offset(cx - w * 0.14, cy);
   517→    final p2 = Offset(cx - w * 0.02, cy + h * 0.12);
   518→    final p3 = Offset(cx + w * 0.20, cy - h * 0.12);
   519→    if (draw <= 0.5) {
   520→      final k = draw * 2;
   521→      canvas.drawLine(p1, Offset(p1.dx + (p2.dx - p1.dx) * k, p1.dy + (p2.dy - p1.dy) * k), checkPaint);
   522→    } else {
   523→      canvas.drawLine(p1, p2, checkPaint);
   524→      final k = (draw - 0.5) * 2;
   525→      canvas.drawLine(p2, Offset(p2.dx + (p3.dx - p2.dx) * k, p2.dy + (p3.dy - p2.dy) * k), checkPaint);
   526→    }
   527→  }
   528→
   529→  @override
   530→  bool shouldRepaint(covariant _ShieldPainter old) =>
   531→      old.ripple != ripple || old.draw != draw || old.primary != primary;
   532→}

   604→class _RestorePainter extends CustomPainter {

   606→  final Color onPrimary;

   609→  final double float; // 0..1 bobbing
   610→  _RestorePainter({

   612→    required this.onPrimary,
   613→    required this.surface,
   614→    required this.spin,
   615→    required this.float,
   616→  });

   618→  @override
   619→  void paint(Canvas canvas, Size size) {
   620→    final cx = size.width / 2;
   621→    final cy = size.height / 2;
   622→    final s = size.width;
   623→    final dy = (float - 0.5) * 6;
   624→
   625→    // Glow
   626→    canvas.drawCircle(
   627→      Offset(cx, cy),
   628→      s * 0.42,
   629→      Paint()
   630→        ..shader = RadialGradient(
   631→          colors: [primary.withOpacity(0.16), primary.withOpacity(0)],
   632→        ).createShader(Rect.fromCircle(center: Offset(cx, cy), radius: s * 0.42)),

   932→class _ProcessingPainter extends CustomPainter {
   933→  final Color primary;
   934→  final Color surface;
   935→  final double spin;
   936→  final double rows;
   937→  _ProcessingPainter({
   938→    required this.primary,
   939→    required this.surface,
   940→    required this.spin,
   941→    required this.rows,
   942→  });
   943→
   944→  @override
   945→  void paint(Canvas canvas, Size size) {
   946→    final cx = size.width / 2;
   947→    final cy = size.height / 2;
   948→    final s = size.width;
   949→
   950→    // Glow
   951→    canvas.drawCircle(
   952→      Offset(cx, cy),
   953→      s * 0.42,
   954→      Paint()
   955→        ..shader = RadialGradient(
   956→          colors: [primary.withOpacity(0.16 + 0.08 * (0.5 + 0.5 * math.sin(spin * 2 * math.pi))), primary.withOpacity(0)],
   957→        ).createShader(Rect.fromCircle(center: Offset(cx, cy), radius: s * 0.42)),
   958→    );
   959→
   960→    // Rotating dashed rings
   961→    canvas.drawArc(
   962→      Rect.fromCircle(center: Offset(cx, cy), radius: s * 0.38),
   963→      spin * 2 * math.pi,
   964→      1.2,
   965→      false,
   966→      Paint()
   967→        ..color = primary.withOpacity(0.5)
   968→        ..style = PaintingStyle.stroke
   969→        ..strokeWidth = 2
   970→        ..strokeCap = StrokeCap.round,
   971→    );
   972→    canvas.drawArc(
   973→      Rect.fromCircle(center: Offset(cx, cy), radius: s * 0.31),
   974→      -spin * 2 * math.pi,
   975→      0.9,
   976→      false,
   977→      Paint()
   978→        ..color = primary.withOpacity(0.35)
   979→        ..style = PaintingStyle.stroke
   980→        ..strokeWidth = 1.5
   981→        ..strokeCap = StrokeCap.round,
   982→    );
   983→
   984→    // Central file card
   985→    final fw = s * 0.32;
   986→    final fh = s * 0.28;
   987→    final fileRect = Rect.fromCenter(center: Offset(cx, cy), width: fw, height: fh);
   988→    canvas.drawRRect(
   989→      RRect.fromRectAndRadius(fileRect, Radius.circular(fw * 0.16)),
   990→      Paint()..color = surface,
   991→    );
   992→    canvas.drawRRect(
   993→      RRect.fromRectAndRadius(fileRect, Radius.circular(fw * 0.16)),
   994→      Paint()
   995→        ..color = primary
   996→        ..style = PaintingStyle.stroke
   997→        ..strokeWidth = 1.5,
   998→    );
   999→
  1000→    // 4 rows being parsed (staggered reveal)
  1001→    const rowCount = 4;
  1002→    for (int i = 0; i < rowCount; i++) {
  1003→      final phase = (rows * rowCount - i) % rowCount;
  1004→      final reveal = phase.clamp(0.0, 1.0);
  1005→      if (reveal <= 0) continue;
  1006→      final y = cy - fh * 0.30 + i * fh * 0.20;
  1007→      final fullW = fw * (0.72 - i * 0.08);
  1008→      final filledW = fullW * reveal;
  1009→      canvas.drawRRect(
  1010→        RRect.fromRectAndRadius(
  1011→          Rect.fromLTWH(cx - fw * 0.36, y, fullW, fh * 0.06),
  1012→          Radius.circular(2),
  1013→        ),
  1014→        Paint()..color = primary.withOpacity(0.15),
  1015→      );
  1016→      if (filledW > 0) {
  1017→        canvas.drawRRect(
  1018→          RRect.fromRectAndRadius(
  1019→            Rect.fromLTWH(cx - fw * 0.36, y, filledW, fh * 0.06),
  1020→            Radius.circular(2),
  1021→          ),
  1022→          Paint()..color = primary.withOpacity(0.85),
  1023→        );
  1024→      }
  1025→    }
  1026→
  1027→    // Parsing particle flowing through
  1028→    final particleT = (spin * 2) % 1.0;
  1029→    final px = cx - fw * 0.40 + fw * 0.80 * particleT;
  1030→    canvas.drawCircle(Offset(px, cy), s * 0.014, Paint()..color = primary);
  1031→  }
  1032→
  1033→  @override
  1034→  bool shouldRepaint(covariant _ProcessingPainter old) =>
  1035→      old.spin != spin || old.rows != rows || old.primary != primary;

  1263→class _RestoreProcessingPainter extends CustomPainter {

  1399→class _CheckCirclePainter extends CustomPainter {
  1400→  final Color primary;
  1401→  final Color onPrimary;
  1402→  final double progress;
  1403→  final double glow;
  1404→  final bool confetti;
  1405→  _CheckCirclePainter({
  1406→    required this.primary,
  1407→    required this.onPrimary,
  1408→    required this.progress,
  1409→    required this.glow,
  1410→    required this.confetti,
  1411→  });
  1412→
  1413→  @override
  1414→  void paint(Canvas canvas, Size size) {
  1415→    final cx = size.width / 2;
  1416→    final cy = size.height / 2;
  1417→    final r = size.width * 0.34;
  1418→
  1419→    // Glow (breathing)
  1420→    canvas.drawCircle(
  1421→      Offset(cx, cy),
  1422→      r * (1.3 + 0.08 * glow),
  1423→      Paint()
  1424→        ..shader = RadialGradient(
  1425→          colors: [primary.withOpacity(0.20 + 0.10 * glow), primary.withOpacity(0)],
  1426→        ).createShader(Rect.fromCircle(center: Offset(cx, cy), radius: r * 1.4)),
  1427→    );
  1428→
  1429→    // Circle (draws in)
  1430→    final circleSweep = (progress * 1.3).clamp(0.0, 1.0);
  1431→    canvas.drawArc(
  1432→      Rect.fromCircle(center: Offset(cx, cy), radius: r),
  1433→      -math.pi / 2,
  1434→      circleSweep * 2 * math.pi,
  1435→      false,
  1436→      Paint()
  1437→        ..color = primary
  1438→        ..style = PaintingStyle.stroke
  1439→        ..strokeWidth = size.width * 0.055
  1440→        ..strokeCap = StrokeCap.round,
  1441→    );
  1442→    // Fill once drawn
  1443→    if (circleSweep >= 1.0) {
  1444→      canvas.drawCircle(Offset(cx, cy), r, Paint()..color = primary);
  1445→    }
  1446→
  1447→    // Check mark (draws after circle)
  1448→    final checkP = (progress * 1.6 - 0.6).clamp(0.0, 1.0);
  1449→    if (checkP > 0) {
  1450→      final p1 = Offset(cx - r * 0.32, cy);
  1451→      final p2 = Offset(cx - r * 0.05, cy + r * 0.30);
  1452→      final p3 = Offset(cx + r * 0.36, cy - r * 0.28);
  1453→      final paint = Paint()
  1454→        ..color = onPrimary
  1455→        ..style = PaintingStyle.stroke
  1456→        ..strokeWidth = size.width * 0.065
  1457→        ..strokeCap = StrokeCap.round
  1458→        ..strokeJoin = StrokeJoin.round;
  1459→      if (checkP <= 0.5) {
  1460→        final k = checkP * 2;
  1461→        canvas.drawLine(p1, Offset(p1.dx + (p2.dx - p1.dx) * k, p1.dy + (p2.dy - p1.dy) * k), paint);
  1462→      } else {
  1463→        canvas.drawLine(p1, p2, paint);
  1464→        final k = (checkP - 0.5) * 2;
  1465→        canvas.drawLine(p2, Offset(p2.dx + (p3.dx - p2.dx) * k, p2.dy + (p3.dy - p2.dy) * k), paint);
  1466→      }
  1467→    }
  1468→
  1469→    // Confetti
  1470→    if (confetti && progress > 0.5) {
  1471→      final rng = math.Random(42);
  1472→      final confettiProgress = (progress - 0.5) * 2;
  1473→      for (int i = 0; i < 22; i++) {
  1474→        final a = rng.nextDouble() * 2 * math.pi;
  1475→        final dist = r * (1.2 + 0.6 * rng.nextDouble()) * confettiProgress;
  1476→        final dx = cx + dist * math.cos(a);
  1477→        final dy = cy + dist * math.sin(a);
  1478→        final sz = size.width * (0.008 + 0.008 * rng.nextDouble());
  1479→        canvas.drawCircle(
  1480→          Offset(dx, dy),
  1481→          sz,
  1482→          Paint()..color = primary.withOpacity(0.7 * (1 - confettiProgress * 0.5)),
  1483→        );
  1484→      }
  1485→    }
  1486→  }
  1487→
  1488→  @override
  1489→  bool shouldRepaint(covariant _CheckCirclePainter old) =>
  1490→      old.progress != progress || old.glow != glow || old.primary != primary;
  1491→}
