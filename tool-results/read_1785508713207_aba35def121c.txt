   137→const STATUS_MESSAGES = [

   146→

   148→    if (!active) {
   149→      setMsgIdx(0);
   150→      return;

   160→      clearTimeout(doneTimer);
   161→    };

   164→  const restoredCount = linkedAnime.filter((a) => a.linked).length + 239;
   165→
   166→  return (
   167→    <div className={`wizard-step wizard-step--v2 ${active ? "wizard-step--active" : ""}`}>
