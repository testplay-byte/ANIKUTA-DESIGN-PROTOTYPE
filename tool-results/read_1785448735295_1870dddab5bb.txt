    25→class AnikutaMainScreen extends StatefulWidget {
    26→  const AnikutaMainScreen({super.key});
    27→
    28→  @override
    29→  State<AnikutaMainScreen> createState() => _AnikutaMainScreenState();
    30→}
    31→
    32→class _AnikutaMainScreenState extends State<AnikutaMainScreen> {
    33→  int _navIndex = 0;
    34→  int? _detailAnimeId;
    35→  int? _watchAnimeId;
    36→  String? _pushedScreen; // 'search', 'history', 'my', 'backup', 'downloads', 'extensions'
    37→
    38→  void _onNavTap(int index) {
    39→    setState(() {
    40→      _navIndex = index;
    41→      _detailAnimeId = null;
    42→      _watchAnimeId = null;
    43→      _pushedScreen = null;
    44→    });
    45→  }
    46→
    47→  void _openAnime(int id) {
    48→    setState(() => _detailAnimeId = id);
    49→  }
    50→
    51→  void _openWatch(int id) {
    52→    setState(() => _watchAnimeId = id);
    53→  }
    54→
    55→  void _openPushed(String screen) {
    56→    setState(() => _pushedScreen = screen);
    57→  }
    58→
    59→  void _pop() {
    60→    setState(() {
    61→      if (_watchAnimeId != null) {
    62→        _watchAnimeId = null;
    63→      } else if (_detailAnimeId != null) {
    64→        _detailAnimeId = null;
    65→      } else if (_pushedScreen != null) {
    66→        _pushedScreen = null;
    67→      }
    68→    });
    69→  }
    70→
    71→  @override
    72→  Widget build(BuildContext context) {
    73→    // Determine what to show
    74→    Widget content;
    75→    bool showNav = true;
    76→
    77→    if (_watchAnimeId != null) {
    78→      content = WatchScreen(animeId: _watchAnimeId!, onBack: _pop);
    79→      showNav = false;
    80→    } else if (_detailAnimeId != null) {
    81→      content = DetailScreen(
    82→        animeId: _detailAnimeId!,
    83→        onBack: _pop,
    84→        onOpenWatch: (id) => setState(() => _watchAnimeId = id),
    85→      );
    86→      showNav = false;
    87→    } else if (_pushedScreen != null) {
    88→      switch (_pushedScreen!) {
    89→        case 'search':
    90→          content = SearchScreen(onBack: _pop, onOpenAnime: _openAnime);
    91→          break;
    92→        case 'history':
    93→          content = HistoryScreen(onBack: _pop, onOpenWatch: _openWatch);
    94→          break;
    95→        case 'my':
    96→          content = const ProfileScreen();
    97→          break;
    98→        case 'backup':
    99→          content = const BackupScreen();
   100→          break;
   101→        case 'downloads':
   102→          content = const DownloadsScreen();
   103→          break;
   104→        case 'extensions':
   105→          content = const ExtensionsScreen();
   106→          break;
   107→        default:
   108→          content = _buildTabScreen();
   109→      }
   110→      showNav = false;
   111→    } else {
   112→      content = _buildTabScreen();
   113→    }
   114→
   115→    return Scaffold(
   116→      body: SafeArea(
   117→        top: false,
   118→        child: Stack(
   119→          children: [
   120→            content,
   121→            if (showNav)
   122→              Positioned(
   123→                left: 0,
   124→                right: 0,
   125→                bottom: 0,
   126→                child: AnikutaBottomNav(
   127→                  currentIndex: _navIndex,
   128→                  onTap: _onNavTap,
   129→                ),
   130→              ),
   131→          ],
   132→        ),
   133→      ),
   134→    );
   135→  }
   136→
   137→  Widget _buildTabScreen() {
   138→    switch (_navIndex) {
   139→      case 0:
   140→        return HomeScreen(
   141→          onOpenAnime: _openAnime,
   142→          onOpenSearch: () => _openPushed('search'),
   143→          onOpenHistory: () => _openPushed('history'),
   144→          onOpenWatch: _openWatch,
   145→        );
   146→      case 1:
   147→        return LibraryScreen(onOpenAnime: _openAnime);
   148→      case 2:
   149→        return UpdatesScreen(onOpenAnime: _openAnime);
   150→      case 3:
   151→        return MoreScreen(
   152→          onOpenBackup: () => _openPushed('backup'),
   153→          onOpenDownloads: () => _openPushed('downloads'),
   154→          onOpenExtensions: () => _openPushed('extensions'),
   155→          onOpenProfile: () => _openPushed('my'),
   156→        );
   157→      default:
   158→        return Container();
   159→    }
   160→  }
   161→}
